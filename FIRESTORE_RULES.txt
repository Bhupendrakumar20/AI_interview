rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // ========================================================
    // HELPER FUNCTIONS
    // ========================================================
    function isAuth() {
      return request.auth != null;
    }
    
    function isUser(uid) {
      return request.auth != null && request.auth.uid == uid;
    }
    
    function isAdmin() {
      return request.auth != null && (request.auth.token.admin == true || request.auth.token.super_admin == true);
    }
    
    // ========================================================
    // USERS SPACE (/users/{userId})
    // ========================================================
    match /users/{userId} {
      // ✅ Profile read/update
      allow read: if isAuth() && (isUser(userId) || isAdmin());
      allow update: if isUser(userId) || isAdmin();
      allow create: if isAuth();
      allow delete: if false; // Immutable accounts
      
      // ✅ 1. Interview Buddy Sessions (Subcollection)
      match /interview_buddy_sessions/{sessionId} {
        allow create: if isUser(userId) && request.resource.data.mode in ['human', 'ai'] && request.resource.data.status == 'created';
        allow read: if isUser(userId) || (isAuth() && request.auth.uid in resource.data.participants) || isAdmin();
        allow update: if isUser(userId) && (request.resource.data.status in ['in-progress', 'completed', 'paused'] || request.resource.data == resource.data);
        allow delete: if isUser(userId);

        // Nested Proctoring Validations
        match /proctoring_validations/{validationId} {
          allow read: if isUser(userId) || (isAuth() && get(/databases/$(database)/documents/users/$(userId)/interview_buddy_sessions/$(sessionId)).data.participants.contains(request.auth.uid));
          allow create: if isAuth() && request.resource.data.userId == request.auth.uid;
          allow update, delete: if false;
        }
      }
      
      // ✅ 2. User Stats (Subcollection)
      match /user_stats/{statType} {
        allow read: if isUser(userId) || isAdmin();
        allow write: if isUser(userId) || isAdmin();
      }
      
      // ✅ 3. Interviews (Subcollection)
      match /interviews/{interviewId} {
        allow read: if isUser(userId) || isAdmin();
        allow create, update, delete: if isUser(userId) || isAdmin();
      }
      
      // ✅ 4. Applications (Subcollection)
      match /applications/{applicationId} {
        allow read: if isUser(userId) || isAdmin();
        allow create: if isUser(userId) && request.resource.data.keys().hasAny(['userId', 'internshipId', 'createdAt']);
        allow update, delete: if isUser(userId) || isAdmin();
      }
      
      // ✅ 5. Feedback (Subcollection)
      match /feedback/{feedbackId} {
        allow read: if isUser(userId) || isAdmin();
        allow write: if isUser(userId) || isAdmin();
      }

      // ✅ 6. Bookmarked Questions (Subcollection)
      match /bookmarked_questions/{questionId} {
        allow read, write: if isUser(userId) || isAdmin();
      }

      // ✅ 7. Room Participants (Subcollection)
      match /room_participants/{participantId} {
        allow read, write: if isUser(userId) || isAdmin();
      }
    }

    // ========================================================
    // PLATFORM SPACE (Global/Shared Root Collections)
    // ========================================================
    
    // DSA Questions
    match /dsa_questions/{questionId} {
      allow read: if true; // Public read
      allow write: if isAdmin();
    }
    
    // DSA Rooms
    match /dsa_rooms/{roomId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update: if isAuth() && (resource.data.createdBy == request.auth.uid || resource.data.host_id == request.auth.uid);
      allow delete: if isAuth() && (resource.data.createdBy == request.auth.uid || resource.data.host_id == request.auth.uid);
    }
    
    // Internships
    match /internships/{internshipId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // Courses & Competitions
    match /courses/{courseId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }
    match /competitions/{compId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // Featured Items
    match /featured_items/{itemId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ========================================================
    // SYSTEM SPACE (Admin-Only /system Subcollections)
    // ========================================================
    match /system/{category} {
      // Allow read/write to system configs/logs only for admins
      allow read, write: if isAdmin();

      // Nested collection rules under /system
      match /{document=**} {
        allow read, write: if isAdmin();
      }
    }

    // ========================================================
    // SECURITY: REJECT ALL OTHER COLLECTIONS
    // ========================================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
